local dragging, dragInput, dragStart, startPos
local function update(input)
    local delta = input.Position - dragStart
    local targetPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    TweenService:Create(mainFrame, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = targetPos}):Play()
end

mainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if input.Position.Y - mainFrame.AbsolutePosition.Y <= 40 then
            dragging = true
            dragStart = input.Position
            startPos = mainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end
end)

mainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        update(input)
    end
end)

local function checkIsRat()
    local char = player.Character
    if not char then return false end
    for _, child in ipairs(char:GetChildren()) do
        if child.Name:lower() == "grabhitbox" then
            return true
        end
    end
    return false
end

local function isInsideVaultVolume(moneyPart)
    local volume = workspace:FindFirstChild("Gameplay") and workspace.Gameplay:FindFirstChild("TheVault") and workspace.Gameplay.TheVault:FindFirstChild("Volume")
    if not volume or not moneyPart then return true end
    local cf, size = volume.CFrame, volume.Size
    local relPos = cf:PointToObjectSpace(moneyPart.Position)
    return math.abs(relPos.X) <= size.X / 2 and math.abs(relPos.Y) <= size.Y / 2 and math.abs(relPos.Z) <= size.Z / 2
end

local b1c = Instance.new("UICorner")

autoFarmBtn.MouseButton1Click:Connect(function()
    autoFarmRunning = not autoFarmRunning
    if autoFarmRunning then
        autoFarmBtn.Text = "Auto farm money: ON"
        autoFarmBtn.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
        
        task.spawn(function()
            local targetTeleCFrame = CFrame.new(-24.5000572, 91.1190414, -154.884506, 1, 0, 0, 0, 1, 0, 0, 0, 1)
            local itemsFolder = workspace:FindFirstChild("Entities") and workspace.Entities:FindFirstChild("Items")
            
            while autoFarmRunning do
                if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
                    task.wait(1)
                    continue
                end
                
                local originalPosition = player.Character.HumanoidRootPart.CFrame

                if not checkIsRat() then
                    showNotification("make sure you are a rat", 3)
                    autoFarmRunning = false
                    autoFarmBtn.Text = "Auto farm money"
                    autoFarmBtn.BackgroundColor3 = Color3.fromRGB(50, 120, 50)
                    break
                end
                
                player.Character.HumanoidRootPart.CFrame = targetTeleCFrame
                
                local existingMoney = {}
                if itemsFolder then
                    for _, child in ipairs(itemsFolder:GetChildren()) do
                        if child.Name == "Money" then
                            existingMoney[child] = true
                        end
                    end
                end
                
                local targetMoney = nil
                if itemsFolder then
                    while autoFarmRunning do
                        local newMoneyFound = nil
                        for _, child in ipairs(itemsFolder:GetChildren()) do
                            if child.Name == "Money" and not existingMoney[child] then
                                if isInsideVaultVolume(child) then
                                    newMoneyFound = child
                                    break
                                end
                            end
                        end
                        
                        if newMoneyFound then
                            targetMoney = newMoneyFound
                            break
                        end
                        task.wait(0.3)
                    end
                end
                
                if not autoFarmRunning then break end
                
                if targetMoney and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = targetMoney.CFrame
                    
                    if remoteEvent then
                        remoteEvent:FireServer("gameplay.grabItemAsAnimal", {
                            targetPart = targetMoney
                        })
                    end
                    task.wait(0.3)
                    
                    if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                        player.Character.HumanoidRootPart.CFrame = originalPosition
                    end
                    
                    task.wait(1)
                    
                    if remoteEvent and autoFarmRunning then
                        remoteEvent:FireServer("gameplay.grabItemAsAnimal")
                    end
                end
                
                task.wait(1)
            end
        end)
    else
        autoFarmBtn.Text = "Auto farm money"
        autoFarmBtn.BackgroundColor3 = Color3.fromRGB(50, 120, 50)
    end
end)

strangeBtn.MouseButton1Click:Connect(function()
    task.spawn(function()
        local itemsFolder = workspace:FindFirstChild("Entities") and workspace.Entities:FindFirstChild("Items")
        if not itemsFolder then return end
        
        local char = player.Character
        if not char or not char:FindFirstChild("HumanoidRootPart") then return end
        
        local targetPotion = nil
        for _, child in ipairs(itemsFolder:GetChildren()) do
            if child.Name:lower():find("potion") or string.match(child.Name, "^1_") then
                targetPotion = child
                break
            end
        end
        
        if not targetPotion then
            for _, child in ipairs(itemsFolder:GetChildren()) do
                if child:IsA("BasePart") or child:IsA("Model") then
                    targetPotion = child
                    break
                end
            end
        end
        
        if not targetPotion then
            showNotification("No Potion found!", 2)
            return
        end
        
        local potionId = targetPotion.Name
        if not string.match(potionId, "^1_") then
            local numbersOnly = string.match(potionId, "%d+")
            if numbersOnly then
                potionId = "1_" .. numbersOnly
            else
                potionId = "1_" .. potionId
            end
        end
        
        if targetPotion:IsA("BasePart") then
            char.HumanoidRootPart.CFrame = targetPotion.CFrame
        elseif targetPotion:IsA("Model") and targetPotion.PrimaryPart then
            char.HumanoidRootPart.CFrame = targetPotion.PrimaryPart.CFrame
        end
        
        if remoteEvent then
            remoteEvent:FireServer("interaction", {
                Object = potionId
            })
        end
    end)
end)

    registerRunning = not registerRunning
    
    local cashRegister = workspace:FindFirstChild("Camera") and workspace.Camera:FindFirstChild("PoIs") and workspace.Camera.PoIs:FindFirstChild("CashRegisterPart")
    if not cashRegister then
        showNotification("CashRegisterPart not found!", 2)
        registerRunning = false
        registerBtn.Text = "Register: OFF"
        registerBtn.BackgroundColor3 = Color3.fromRGB(180, 120, 40)
        return
    end
    
    if registerRunning then
        if cashRegister:IsA("BasePart") then
            registerOriginalCF = cashRegister.CFrame
        elseif cashRegister:IsA("Model") and cashRegister.PrimaryPart then
            registerOriginalCF = cashRegister.PrimaryPart.CFrame
        end
        
        registerBtn.Text = "Register: ON"
        registerBtn.BackgroundColor3 = Color3.fromRGB(220, 150, 40)
        
        registerConnection = RunService.RenderStepped:Connect(function()
            local char = player.Character
            if char and char:FindFirstChild("HumanoidRootPart") and cashRegister and cashRegister.Parent then
                local frontCF = char.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
                if cashRegister:IsA("BasePart") then
                    cashRegister.CFrame = frontCF
                elseif cashRegister:IsA("Model") and cashRegister.PrimaryPart then
                    cashRegister:SetPrimaryPartCFrame(frontCF)
                end
            end
        end)
    else
        registerBtn.Text = "Register: OFF"
        registerBtn.BackgroundColor3 = Color3.fromRGB(180, 120, 40)
        
        if registerConnection then
            registerConnection:Disconnect()
            registerConnection = nil
        end
        
        if registerOriginalCF then
            if cashRegister:IsA("BasePart") then
                cashRegister.CFrame = registerOriginalCF
            elseif cashRegister:IsA("Model") and cashRegister.PrimaryPart then
                cashRegister:SetPrimaryPartCFrame(registerOriginalCF)
            end
        end
    end

-- Hàm kiểm tra trạng thái Rat
local function checkIsRat()
    local gameplayVal = player:FindFirstChild("Gameplay")
    if gameplayVal and gameplayVal:IsA("ValueBase") then
        return tostring(gameplayVal.Value):lower() == "rat"
    elseif player:FindFirstChild("Gameplay") then
        -- Trường hợp là thuộc tính hoặc object khác
        return tostring(player.Gameplay) == "rat"
    end
    return false
end

-- Hàm kiểm tra xem Part Money có nằm trong Volume (TheVault.Volume) hay không
local function isInsideVaultVolume(moneyPart)
    local volume = workspace:FindFirstChild("Gameplay") and workspace.Gameplay:FindFirstChild("TheVault") and workspace.Gameplay.TheVault:FindFirstChild("Volume")
    if not volume or not moneyPart then return true end -- Nếu không tìm thấy volume thì bỏ qua bước check này để tránh kẹt
    
    local cf, size = volume.CFrame, volume.Size
    local relPos = cf:PointToObjectSpace(moneyPart.Position)
    return math.abs(relPos.X) <= size.X / 2 and math.abs(relPos.Y) <= size.Y / 2 and math.abs(relPos.Z) <= size.Z / 2
end

-- NÚT 1: Auto farm money
--------------------------------------------------------------------------------
local autoFarmRunning = false
local autoFarmBtn = Instance.new("TextButton")
autoFarmBtn.Size = UDim2.new(1, 0, 0, 45)
autoFarmBtn.BackgroundColor3 = Color3.fromRGB(50, 120, 50)
autoFarmBtn.Text = "Auto farm money: OFF"
autoFarmBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
autoFarmBtn.TextSize = 15
autoFarmBtn.Font = Enum.Font.SourceSansBold
autoFarmBtn.Parent = container

local b1c = Instance.new("UICorner")
b1c.CornerRadius = UDim.new(0, 6)
b1c.Parent = autoFarmBtn

autoFarmBtn.MouseButton1Click:Connect(function()
    autoFarmRunning = not autoFarmRunning
    if autoFarmRunning then
        autoFarmBtn.Text = "Auto farm money: ON"
        autoFarmBtn.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
        
        task.spawn(function()
            local startCFrame = CFrame.new(-24.5000572, 91.1190414, -154.884506, 1, 0, 0, 0, 1, 0, 0, 0, 1)
            
            while autoFarmRunning do
                -- Kiểm tra rat trước khi làm gì
                if not checkIsRat() then
                    showNotification("make sure you are a rat", 3)
                    autoFarmRunning = false
                    autoFarmBtn.Text = "Auto farm money: OFF"
                    autoFarmBtn.BackgroundColor3 = Color3.fromRGB(50, 120, 50)
                    break
                end
                
                -- Teleport đến vị trí bắt đầu
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = startCFrame
                end
                
                -- Đợi part Money xuất hiện trong workspace.Entities.Items và nằm trong Volume
                local moneyPart = nil
                local itemsFolder = workspace:FindFirstChild("Entities") and workspace.Entities:FindFirstChild("Items")
                
                if itemsFolder then
                    -- Lắng nghe hoặc quét tìm part Money
                    while autoFarmRunning do
                        moneyPart = itemsFolder:FindFirstChild("Money")
                        if moneyPart and isInsideVaultVolume(moneyPart) then
                            break
                        end
                        task.wait(0.5)
                    end
                end
                
                if not autoFarmRunning then break end
                
                -- Teleport tới part Money và kích hoạt RemoteEvent liên tục (nhưng có độ trễ nhỏ)
                if moneyPart and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = moneyPart.CFrame
                    
                    while autoFarmRunning and moneyPart and moneyPart.Parent do
                        if remoteEvent then
                            remoteEvent:FireServer("gameplay.grabItemAsAnimal", {
                                targetPart = moneyPart
                            })
                        end
                        
                        -- Kiểm tra nếu Money đã nằm trong Character
                        if player.Character:FindFirstChild("Money") or (moneyPart.Parent == player.Character) then
                            break
                        end
                        task.wait(0.3)
                    end
                    
                    -- Teleport quay lại chỗ nút bật Auto farm money và nhả tiền ra
                    if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                        player.Character.HumanoidRootPart.CFrame = startCFrame
                    end
                    
                    if remoteEvent then
                        remoteEvent:FireServer("gameplay.grabItemAsAnimal", {})
                    end
                end
                
                task.wait(1)
            end
        end)
    else
        autoFarmBtn.Text = "Auto farm money: OFF"
        autoFarmBtn.BackgroundColor3 = Color3.fromRGB(50, 120, 50)
    end
end)

-- NÚT 3: Get strange point
--------------------------------------------------------------------------------
local strangeBtn = Instance.new("TextButton")
strangeBtn.Size = UDim2.new(1, 0, 0, 45)
strangeBtn.BackgroundColor3 = Color3.fromRGB(150, 70, 150)
strangeBtn.Text = "Get Strange Point"
strangeBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
strangeBtn.TextSize = 15
strangeBtn.Font = Enum.Font.SourceSansBold
strangeBtn.Parent = container

local b3c = Instance.new("UICorner")
b3c.CornerRadius = UDim.new(0, 6)
b3c.Parent = strangeBtn

strangeBtn.MouseButton1Click:Connect(function()
    task.spawn(function()
        local itemsFolder = workspace:FindFirstChild("Entities") and workspace.Entities:FindFirstChild("Items")
        if not itemsFolder then return end
        
        -- Lấy vị trí ban đầu của character
        local char = player.Character
        if not char or not char:FindFirstChild("HumanoidRootPart") then return end
        local originalPos = char.HumanoidRootPart.CFrame
        
        -- Tìm point đầu tiên trong itemsFolder (lọc tìm object phù hợp hoặc Potion/Point)
        local targetPoint = nil
        for _, child in ipairs(itemsFolder:GetChildren()) do
            if child:IsA("BasePart") or child:IsA("Model") then
                targetPoint = child
                break
            end
        end
        
        if not targetPoint then return end
        
        local pointId = targetPoint.Name
        -- Trích xuất hoặc giữ nguyên định dạng id kiểu "1_" nếu có
        if not string.match(pointId, "^1_") then
            pointId = "1_" .. pointId
        end
        
        -- Đưa point di chuyển mượt mà trước mặt character (giả lập anchored nhưng không anchored)
        local connection
        connection = RunService.RenderStepped:Connect(function()
            if char and char:FindFirstChild("HumanoidRootPart") and targetPoint then
                local targetCFrame = char.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
                if targetPoint:IsA("Model") and targetPoint.PrimaryPart then
                    targetPoint:SetPrimaryPartCFrame(targetCFrame)
                elseif targetPoint:IsA("BasePart") then
                    targetPoint.CFrame = targetCFrame
                end
            end
        end)
        
        -- Kiểm tra sau 5 giây nếu chưa biến thành rat
        task.delay(5, function()
            if not checkIsRat() then
                if connection then connection:Disconnect() end
                -- Teleport mình đến vị trí point và trả lại vị trí point
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local currentPointPos = originalPos
                    if targetPoint:IsA("BasePart") then
                        currentPointPos = targetPoint.CFrame
                    end
                    char.HumanoidRootPart.CFrame = currentPointPos
                end
            end
        end)
        
        -- Kích hoạt event uống Potion
        if remoteEvent then
            remoteEvent:FireServer("interaction", {
                Object = pointId -- Ví dụ: GetNil("Potion", "1_578690") tương ứng
            })
        end
        
        -- Ngắt kết nối sau một khoảng thời gian hoàn thành
        task.wait(3)
        if connection then connection:Disconnect() end
    end)
end)
